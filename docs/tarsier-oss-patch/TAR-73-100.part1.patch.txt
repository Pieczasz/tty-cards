+func headerlessMessageAssign(pass *analysis.Pass, stmt ast.Stmt) (string, *ast.CompositeLit, bool) {
+	as, ok := stmt.(*ast.AssignStmt)
+	if !ok || (as.Tok != token.DEFINE && as.Tok != token.ASSIGN) {
+		return "", nil, false
+	}
+	if len(as.Lhs) != 1 || len(as.Rhs) != 1 {
+		return "", nil, false
+	}
+	id, ok := as.Lhs[0].(*ast.Ident)
+	if !ok || id.Name == "_" {
+		return "", nil, false
+	}
+	lit, ok := unwrapMessageLit(as.Rhs[0])
+	if !ok || !isMessageLit(pass, lit) || hasHeaders(lit) {
+		return "", nil, false
+	}
+	return id.Name, lit, true
+}

+func produceUsesIdent(stmt ast.Stmt, name string) bool {
+	var found bool
+	ast.Inspect(stmt, func(n ast.Node) bool {
+		call, ok := n.(*ast.CallExpr)
+		if !ok || !isProduceCall(call) {
+			return true
+		}
+		for _, arg := range call.Args {
+			id, ok := unwrapIdent(arg)
+			if ok && id.Name == name {
+				found = true
+				return false
+			}
+		}
+		return true
+	})
+	return found
+}
+
+func unwrapMessageLit(expr ast.Expr) (*ast.CompositeLit, bool) {
+	switch e := expr.(type) {
+	case *ast.CompositeLit:
+		return e, true
+	case *ast.UnaryExpr:
+		if e.Op == token.AND {
+			if lit, ok := e.X.(*ast.CompositeLit); ok {
+				return lit, true
+			}
+		}
+	}
+	return nil, false
+}
+
+func unwrapIdent(expr ast.Expr) (*ast.Ident, bool) {
+	switch e := expr.(type) {
+	case *ast.Ident:
+		return e, true
+	case *ast.UnaryExpr:
+		if e.Op == token.AND || e.Op == token.MUL {
+			if id, ok := e.X.(*ast.Ident); ok {
+				return id, true
+			}
+		}
+	}
+	return nil, false
+}
